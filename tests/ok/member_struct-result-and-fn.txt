struct Point
{
    i32 x
    i32 y
    fn moved := (i32 dx) -> Point
    {
        Point q{x + dx, y}
        exit q
    }
    fn d2 := () -> i32
    {
        i32 r{x * x + y * y}
        exit r
    }
}
struct Circle
{
    Point center
    i32 radius
    fn far := () -> i32
    {
        i32 r{center.d2() + radius}
        exit r
    }
}
fn measure := (Point p) -> i32
{
    exit p.d2()
}
Point p{3, 4}
Point q{p.moved(3)}
Circle c{q, 1}
i32 r{measure(q) + c.far()}
exit r
