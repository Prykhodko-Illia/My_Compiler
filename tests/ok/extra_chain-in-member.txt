struct Point
{
    i32 mut x
    i32 mut y
    fn moved := (i32 dx) -> Point
    {
        Point q{x + dx, y}
        exit q
    }
    fn d2 := () -> i32
    {
        i32 r{x * x + y * y}
        exit r
    }
    fn far := () -> i32
    {
        i32 r{moved(2).d2() + moved(1).moved(1).x}
        exit r
    }
}
struct Circle
{
    Point center
    i32 radius
    fn grown := (i32 dr) -> Circle
    {
        Circle g{center, radius + dr}
        exit g
    }
}
Point p{1, 2}
Circle c{p, 2}
print(c.grown(1).center)
print(c.grown(1).grown(2).radius)
print(c.grown(1).center.moved(5))
i32 x{c.grown(1).center.x}
i32 r{p.far() + x}
exit r
