fn first := (i32 v) -> i32
{
    fn helper := [v]() -> i32
    {
        i32 r{v + 1}
        exit r
    }
    exit helper()
}
fn second := (i64 v) -> i64
{
    fn helper := [v]() -> i64
    {
        i64 r{v * 2}
        exit r
    }
    exit helper()
}
print(first(1))
print(second(4000000000))
exit first(41)
